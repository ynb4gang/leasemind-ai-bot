import tkinter as tk
from tkinter import ttk
from tkinter import colorchooser
import requests

ip = '172.20.10.10'

color = "#FFFFFF"


root = tk.Tk()

root.title("RGB LED Strip Controller")
root.resizable(False, False)
root.geometry("1280x800")


delay = tk.IntVar(value=99)
forward = tk.IntVar(value=1)


def deLay(value):
    labeldelay.config(
        text="DELAY: " + str(delay.get())
    )


def clear():
    global ip, forward, color, delay

    try:
        params = (
            str(forward.get())
            + color[1:]
            + f"{delay.get():02d}"
            + "1"
        )

        string = "http://" + ip + "/clear?params=" + params

        requests.post(string)

        print(string)

    except Exception as E:
        print(f"** Error {E} **")


def setcolor():
    global ip, forward, color, delay

    try:
        params = (
            str(forward.get())
            + color[1:]
            + f"{delay.get():02d}"
            + "0"
        )

        string = "http://" + ip + "/fill?params=" + params

        requests.post(string)

        print(string)

    except Exception as E:
        print(f"** Error {E} **")


def rainbow():
    global ip, forward, color, delay

    try:
        params = (
            str(forward.get())
            + color[1:]
            + f"{delay.get():02d}"
            + "2"
        )

        string = "http://" + ip + "/rainbow?params=" + params

        requests.post(string)

        print(string)

    except Exception as E:
        print(f"** Error {E} **")


def select():
    global color

    try:
        color_code = colorchooser.askcolor(
            title="Select a color"
        )

        if color_code[1]:
            color = color_code[1]

            labelcolor.config(
                text="COLOR: " + color,
                fg=color
            )

            print(color[1:])

    except Exception as E:
        print(f"** Error {E} **")


buttons_frame = tk.Frame(root)
buttons_frame.pack(pady=10)


button_clear = tk.Button(
    buttons_frame,
    text="CLEAR",
    font=("Calibri", 48),
    command=clear
)

button_clear.grid(
    row=7,
    column=0,
    padx=1
)


button_setcolor = tk.Button(
    buttons_frame,
    text="SET COLOR",
    font=("Calibri", 48),
    command=setcolor
)

button_setcolor.grid(
    row=7,
    column=1,
    padx=1
)


button_rainbow = tk.Button(
    buttons_frame,
    text="RAINBOW",
    font=("Calibri", 48),
    command=rainbow
)

button_rainbow.grid(
    row=7,
    column=2,
    padx=1
)


button_select = tk.Button(
    buttons_frame,
    text="SELECT A COLOR",
    anchor=tk.CENTER,
    font=("Calibri", 48),
    command=select
)

button_select.grid(
    row=2,
    column=1,
    padx=1
)


labelcolor = tk.Label(
    buttons_frame,
    text="COLOR: NONE",
    font=("Calibri", 48)
)

labelcolor.grid(
    row=1,
    column=1,
    padx=1
)


labeldelay = tk.Label(
    buttons_frame,
    text="DELAY: 99",
    font=("Calibri", 48)
)

labeldelay.grid(
    row=3,
    column=1,
    padx=1
)


labeldir = tk.Label(
    buttons_frame,
    text="DIRECTION:",
    font=("Calibri", 48)
)

labeldir.grid(
    row=5,
    column=1,
    padx=1
)


sliderdelay = ttk.Scale(
    buttons_frame,
    from_=1,
    to=99,
    length=400,
    command=deLay,
    variable=delay
)

sliderdelay.grid(
    row=4,
    column=1,
    padx=1
)


dirradio1 = tk.Radiobutton(
    buttons_frame,
    text="FOR",
    value=1,
    font=("Calibri", 48),
    variable=forward
)

dirradio1.grid(
    row=6,
    column=0
)


dirradio2 = tk.Radiobutton(
    buttons_frame,
    text="BACK",
    value=2,
    font=("Calibri", 48),
    variable=forward
)

dirradio2.grid(
    row=6,
    column=2
)


try:
    root.mainloop()

except KeyboardInterrupt:
    root.destroy()
